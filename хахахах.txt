local function httpGet(url)
	if type(game.HttpGet) == "function" then
		local ok, res = pcall(game.HttpGet, game, url)
		if ok and type(res) == "string" then return res end
	end
	if type(httpget) == "function" then
		local ok, res = pcall(httpget, url)
		if ok and type(res) == "string" then return res end
	end
	if type(request) == "function" then
		local ok, res = pcall(request, { Url = url, Method = "GET" })
		if ok and type(res) == "table" then return res.Body end
	end
	return nil
end

local function freeDrawing(obj)
	if not obj then return end
	if not pcall(function() obj:Remove() end) then
		pcall(function() obj:Destroy() end)
	end
end

---------------------------------------------------------------- helpers
local function isKO(char, hum)
	if hum and hum.Health <= 0 then return true end
	if char:GetAttribute("KO") or char:GetAttribute("Knocked") or char:GetAttribute("Ragdoll") then return true end
	local fx = char:FindFirstChild("BodyEffects")
	local ko = fx and fx:FindFirstChild("K.O")
	if ko and ko.Value then return true end
	return false
end

local function screenDist(pos)
	local v, on = cam:WorldToViewportPoint(pos)
	if not on or v.Z <= 0 then return nil end
	return (Vector2.new(v.X, v.Y) - UIS:GetMouseLocation()).Magnitude
end

local function visibleFrom(origin, part, targetChar)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { lp.Character, targetChar }
	return Workspace:Raycast(origin, part.Position - origin, params) == nil
end

local function getRole(plr)
	return roles[plr.UserId] or "Neutral"
end

local function isIgnored(plr)
	return getRole(plr) == "Friend"
end

end

local visClear
do
	local visDrawingOk = (type(Drawing) == "table") or (type(Drawing) == "userdata")
	local visFolder = Instance.new("Folder")
	visFolder.Name = "F_" .. TAG
	pcall(function() visFolder.Parent = guiContainer() end)
	local visObjects = {}

	local function visDraw(class, props)
		if not visDrawingOk then return nil end
		local ok, obj = pcall(Drawing.new, class)
		if not ok or not obj then return nil end
		for k, v in pairs(props) do obj[k] = v end
		obj.Visible = false
		return obj
	end

	local function visCreate(plr)
		if visObjects[plr] then return end
		local hl = Instance.new("Highlight")
		hl.Name = plr.Name
		hl.FillTransparency = 0.5
		hl.OutlineTransparency = 0
		hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		hl.Enabled = false
		hl.Parent = visFolder
		local o = { hl = hl }
		if visDrawingOk then
			o.box    = visDraw("Square", { Thickness = 1.5, Filled = false })
			o.name   = visDraw("Text", { Center = true, Outline = true })
			o.dist   = visDraw("Text", { Center = true, Outline = true })
			o.hpBack = visDraw("Square", { Filled = true, Color = Color3.new(0, 0, 0) })
			o.hpFill = visDraw("Square", { Filled = true })
			o.tracer = visDraw("Line", { Thickness = 1 })
		end
		visObjects[plr] = o
	end

	local function visHide(o)
		for k, v in pairs(o) do
			if k ~= "hl" and v then v.Visible = false end
		end
		if o.hl then o.hl.Enabled = false end
	end

	local function visDestroy(plr)
		local o = visObjects[plr]
		if not o then return end
		for k, v in pairs(o) do
			if k == "hl" then
				pcall(function() v:Destroy() end)
			elseif v then
				freeDrawing(v)
			end
		end
		visObjects[plr] = nil
	end

	function visClear()
		for plr in pairs(visObjects) do
			visDestroy(plr)
		end
		if visFolder then pcall(function() visFolder:Destroy() end) end
	end

	local function visScreenBounds(char)
		local cf, size = char:GetBoundingBox()
		local minX, minY = math.huge, math.huge
		local maxX, maxY = -math.huge, -math.huge
		for x = -1, 1, 2 do
			for y = -1, 1, 2 do
				for z = -1, 1, 2 do
					local world = cf:PointToWorldSpace(Vector3.new(x, y, z) * size / 2)
					local v = cam:WorldToViewportPoint(world)
					if v.Z <= 0 then return nil end
					minX = math.min(minX, v.X)
					minY = math.min(minY, v.Y)
					maxX = math.max(maxX, v.X)
					maxY = math.max(maxY, v.Y)
				end
			end
		end
		return minX, minY, maxX, maxY
	end

	local function visUpdate(plr, o)
		local char = plr.Character
		if not char then return visHide(o) end
		local hum  = char:FindFirstChildOfClass("Humanoid")
		local root = char:FindFirstChild("HumanoidRootPart")
		if not (hum and root and hum.Health > 0) then
			return visHide(o)
		end
		if isIgnored(plr) then
			return visHide(o)
		end
		if cfg.visTeamCheck and plr.Team ~= nil and plr.Team == lp.Team then
			return visHide(o)
		end
		local dist = (root.Position - cam.CFrame.Position).Magnitude
		if cfg.visMaxDist > 0 and dist > cfg.visMaxDist then
			return visHide(o)
		end
		o.hl.Adornee = char
		o.hl.FillColor = cfg.visChamsFill
		o.hl.OutlineColor = cfg.visChamsOutline
		o.hl.Enabled = cfg.visChams
		if not visDrawingOk then return end
		if not o.box then return end
		local minX, minY, maxX, maxY = visScreenBounds(char)
		if not minX then
			for k, v in pairs(o) do
				if k ~= "hl" and v then v.Visible = false end
			end
			return
		end
		local w, h = maxX - minX, maxY - minY
		local cx = (minX + maxX) / 2
		o.box.Visible = cfg.visBox
		if cfg.visBox then
			o.box.Color = cfg.visBoxColor
			o.box.Position = Vector2.new(minX, minY)
			o.box.Size = Vector2.new(w, h)
		end
		o.name.Visible = cfg.visName
		if cfg.visName then
			o.name.Text = plr.DisplayName ~= plr.Name and (plr.DisplayName .. " (" .. plr.Name .. ")") or plr.Name
			o.name.Size = cfg.visTextSize
			o.name.Color = cfg.visNameColor
			o.name.Position = Vector2.new(cx, minY - cfg.visTextSize - 3)
		end
		o.dist.Visible = cfg.visDistance
		if cfg.visDistance then
			o.dist.Text = string.format("%d studs", math.floor(dist))
			o.dist.Size = math.max(cfg.visTextSize - 1, 8)
			o.dist.Color = cfg.visNameColor
			o.dist.Position = Vector2.new(cx, maxY + 2)
		end
		o.hpBack.Visible = cfg.visHealth
		o.hpFill.Visible = cfg.visHealth
		if cfg.visHealth then
			local ratio = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
			local fillH = h * ratio
			o.hpBack.Position = Vector2.new(minX - 6, minY - 1)
			o.hpBack.Size = Vector2.new(5, h + 2)
			o.hpFill.Position = Vector2.new(minX - 5, minY + h - fillH)
			o.hpFill.Size = Vector2.new(3, fillH)
			o.hpFill.Color = Color3.fromRGB(255, 60, 60):Lerp(Color3.fromRGB(60, 255, 90), ratio)
		end
		o.tracer.Visible = cfg.visTracers
		if cfg.visTracers then
			o.tracer.Color = cfg.visTracerColor
			o.tracer.From = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y)
			o.tracer.To = Vector2.new(cx, maxY)
		end
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= lp then visCreate(plr) end
	end
	bind(Players.PlayerAdded, function(plr)
		if plr ~= lp then visCreate(plr) end
	end)
	bind(Players.PlayerRemoving, visDestroy)

	local visWasOn = false
	bind(RS.RenderStepped, function()
		cam = Workspace.CurrentCamera or cam
		local on = cfg.vis
		if on then
			for plr, o in pairs(visObjects) do
				visUpdate(plr, o)
			end
		elseif visWasOn then
			for _, o in pairs(visObjects) do
				visHide(o)
			end
		end
		visWasOn = on
	end)
end -- Visuals (ESP)

---------------------------------------------------------------- Silent Aim
local silentSupported, silentState, applySilent, silentDisableHooks, silentCircle, bulletTpShot, tpRestore
silentState = { part = nil, char = nil, aim = nil }
do
	silentSupported = type(hookmetamethod) == "function"
		and type(getnamecallmethod) == "function"
		and type(checkcaller) == "function"
	local silentOldNamecall
	local silentHooked = false
	local silentTargetPart, silentTargetChar, silentAim = nil, nil, nil
	local silentRollT, silentRollOk = 0, true
	local silentUnpack = table.unpack or unpack
	local silentClosure = type(newcclosure) == "function" and newcclosure or function(f) return f end

	local function silentShouldRedirect()
		if not (cfg.silent and silentAim) then return false end
		if cfg.silentShootOnly and not cfg.trigSilent
			and not UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
			return false
		end
		if cfg.silentHitChance < 100 then
			local now = os.clock()
			if now - silentRollT > 0.08 then
				silentRollT = now
				silentRollOk = math.random(1, 100) <= cfg.silentHitChance
			end
			if not silentRollOk then return false end
		end
		return true
	end

	local SILENT_HEAD_PARTS = { "Head" }
	local SILENT_BODY_PARTS = { "UpperTorso", "Torso", "LowerTorso", "HumanoidRootPart" }
	local SILENT_CLOSEST_PARTS = {
		"Head", "UpperTorso", "LowerTorso", "HumanoidRootPart", "Torso",
		"LeftUpperArm", "RightUpperArm", "LeftUpperLeg", "RightUpperLeg",
		"Left Arm", "Right Arm", "Left Leg", "Right Leg",
	}

	local function silentFirstPart(char, names)
		for _, name in ipairs(names) do
			local p = char:FindFirstChild(name)
			if p and p:IsA("BasePart") then return p end
		end
		return nil
	end

	local function silentPickPart(char)
		local mode = cfg.silentPart
		if mode == "Random" then
			mode = math.random(1, 2) == 1 and "Head" or "Body"
		end
		if mode == "Head" then
			return silentFirstPart(char, SILENT_HEAD_PARTS)
		end
		if mode == "Body" then
			return silentFirstPart(char, SILENT_BODY_PARTS)
		end
		if mode ~= "Closest body part" then
			local p = char:FindFirstChild(mode)
			if p and p:IsA("BasePart") then return p end
		end
		local best, bestD = nil, math.huge
		for _, name in ipairs(SILENT_CLOSEST_PARTS) do
			local p = char:FindFirstChild(name)
			if p and p:IsA("BasePart") then
				local d = screenDist(p.Position)
				if d and d < bestD then best, bestD = p, d end
			end
		end
		return best
	end

	local function silentFindTarget()
		local myChar = lp.Character
		local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
		if not myRoot then return nil end
		local best, bestD = nil, cfg.silentFov
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= lp and not isIgnored(plr) then
				local char = plr.Character
				local hum = char and char:FindFirstChildOfClass("Humanoid")
				local root = char and char:FindFirstChild("HumanoidRootPart")
				if char and hum and root and hum.Health > 0
					and not (cfg.silentTeamCheck and plr.Team ~= nil and plr.Team == lp.Team)
					and not (cfg.silentSkipKO and isKO(char, hum))
					and not (cfg.silentMaxDist > 0 and (root.Position - myRoot.Position).Magnitude > cfg.silentMaxDist) then
					local part = silentPickPart(char)
					local d = part and screenDist(part.Position)
					if d and d < bestD and (not cfg.silentWallCheck or visibleFrom(cam.CFrame.Position, part, char)) then
						best, bestD = part, d
					end
				end
			end
		end
		return best
	end

	pcall(function()
		silentCircle = Drawing.new("Circle")
		silentCircle.Thickness = 1.5
		silentCircle.NumSides = 64
		silentCircle.Filled = false
		silentCircle.Color = Color3.fromRGB(170, 90, 255)
		silentCircle.Visible = false
	end)

	local function silentEnableHooks()
		if silentHooked or not silentSupported then return end
		silentOldNamecall = hookmetamethod(game, "__namecall", silentClosure(function(self, ...)
			if getnamecallmethod() == "Raycast" and self == Workspace
				and not checkcaller() and silentShouldRedirect() then
				local args = { ... }
				local origin, dir = args[1], args[2]
				if typeof(origin) == "Vector3" and typeof(dir) == "Vector3" then
					local delta = silentAim - origin
					if delta.Magnitude > 0.001 then
						args[2] = delta.Unit * dir.Magnitude
						return silentOldNamecall(self, silentUnpack(args))
					end
				end
			end
			return silentOldNamecall(self, ...)
		end))
		silentHooked = true
	end

	function silentDisableHooks()
		silentHooked = false
		local namecall = silentOldNamecall
		silentOldNamecall = nil
		if not namecall then return end
		pcall(function()
			hookmetamethod(game, "__namecall", namecall)
		end)
	end

	function applySilent()
		if cfg.silent and silentSupported and cfg.silentMethod == "Raycast" then
			silentEnableHooks()
		else
			silentDisableHooks()
		end
	end

---------------------------------------------------------------- Bullet teleport
	local TP_GRIP = CFrame.new(0, -1, 0, 1, 0, 0, 0, 0, 1, 0, -1, 0)
	local tpBusy = false

	local function tpGrip(origin, target)
		return (origin * TP_GRIP):ToObjectSpace(target):inverse()
	end

	local function tpFreezeGrip(tool)
		if type(getconnections) ~= "function" then return end
		pcall(function()
			for _, c in ipairs(getconnections(tool:GetPropertyChangedSignal("Grip"))) do
				pcall(function() c:Disable() end)
			end
		end)
	end

	bulletTpShot = function(tool, char)
		if tpBusy or not tool or not char or not char.Parent then return false end
		local me = lp.Character
		local bp = lp:FindFirstChild("Backpack")
		local hand = me and (me:FindFirstChild("RightHand") or me:FindFirstChild("Right Arm"))
		local part = char:FindFirstChild(cfg.tpPart) or char:FindFirstChild("HumanoidRootPart")
		if not (me and bp and hand and part and tool.Parent == me) then return false end
		tpBusy = true
		tpFreezeGrip(tool)
		task.spawn(function()
			local grip = tool.Grip
			pcall(function()
				tool.Parent = bp
				tool.Grip = tpGrip(hand.CFrame, part.CFrame)
				tool.Parent = me
			end)
			RS.RenderStepped:Wait()
			pcall(function() tool:Activate() end)
			RS.RenderStepped:Wait()
			pcall(function()
				tool.Parent = bp
				tool.Grip = grip
				tool.Parent = me
			end)
			tpBusy = false
		end)
		return true
	end

	local tpHoldTool, tpHoldChar, tpHoldGrip, tpHoldPos = nil, nil, nil, nil
	tpRestore = function()
		local t, g = tpHoldTool, tpHoldGrip
		tpHoldTool, tpHoldChar, tpHoldGrip, tpHoldPos = nil, nil, nil, nil
		if not (t and t.Parent and g) then return end
		local me = lp.Character
		local bp = lp:FindFirstChild("Backpack")
		if not (me and bp and t.Parent == me) then
			pcall(function() t.Grip = g end)
			return
		end
		pcall(function()
			t.Parent = bp
			t.Grip = g
			t.Parent = me
		end)
	end

	local function tpTurbo()
		if not (cfg.tpTurbo and cfg.silent and cfg.silentMethod == "Bullet teleport") then
			if tpHoldTool then tpRestore() end
			return
		end
		local me = lp.Character
		local bp = lp:FindFirstChild("Backpack")
		local tool = me and me:FindFirstChildOfClass("Tool")
		local char = silentState.char
		local hand = me and (me:FindFirstChild("RightHand") or me:FindFirstChild("Right Arm"))
		local part = char and (char:FindFirstChild(cfg.tpPart) or char:FindFirstChild("HumanoidRootPart"))
		if not (tool and bp and hand and part) then
			if tpHoldTool then tpRestore() end
			return
		end
		if tpHoldTool == tool and tpHoldChar == char and tpHoldPos
			and (tpHoldPos - part.Position).Magnitude < 2 then
			return
		end
		if tpHoldTool and tpHoldTool ~= tool then tpRestore() end
		if tpHoldTool ~= tool then
			tpFreezeGrip(tool)
			tpHoldGrip = tool.Grip
		end
		local ok = pcall(function()
			tool.Parent = bp
			tool.Grip = tpGrip(hand.CFrame, part.CFrame)
			tool.Parent = me
		end)
		if ok then
			tpHoldTool, tpHoldChar, tpHoldPos = tool, char, part.Position
		end
	end

	bind(RS.RenderStepped, function()
		cam = Workspace.CurrentCamera or cam
		if cfg.silent then
			silentTargetPart = silentFindTarget()
			if silentTargetPart then
				local p = silentTargetPart.Position
				if cfg.silentPredict > 0 then
					p = p + silentTargetPart.AssemblyLinearVelocity * cfg.silentPredict
				end
				if cfg.silentJitter > 0 then
					local j = cfg.silentJitter
					p = p + Vector3.new(
						(math.random() * 2 - 1) * j,
						(math.random() * 2 - 1) * j,
						(math.random() * 2 - 1) * j
					)
				end
				silentAim = p
				silentTargetChar = silentTargetPart.Parent
			else
				silentAim, silentTargetChar = nil, nil
			end
		else
			silentTargetPart, silentTargetChar, silentAim = nil, nil, nil
		end
		silentState.part, silentState.char, silentState.aim = silentTargetPart, silentTargetChar, silentAim
		tpTurbo()
		if silentCircle then
			local show = cfg.silent and cfg.silentFovShow
			silentCircle.Visible = show
			if show then
				silentCircle.Radius = cfg.silentFov
				silentCircle.Position = UIS:GetMouseLocation()
			end
		end
	end)
end -- Silent Aim

---------------------------------------------------------------- triggerbot (ОПТИМИЗИРОВАНО: 2 потока)
local getTool, getTriggerDist, updateTrigger, disconnectTrigger
local trigApi = {}
do
	local triggerRunning = false
	local mouse = lp:GetMouse()
	local cTool, cAmmo
	local cachedChar, cachedTool, cachedAmmo

	function trigApi.ko(pl)
		local root = Workspace:FindFirstChild("Players") and Workspace.Players:FindFirstChild(pl.Name) or pl.Character
		if not root then return false end
		local be = root:FindFirstChild("BodyEffects")
		if not be then return false end
		local ko = be:FindFirstChild("K.O") or be:FindFirstChild("K.O.")
		return (ko and ko:IsA("BoolValue") and ko.Value) or false
	end

	function trigApi.wall(part)
		local myRoot = cachedChar and cachedChar:FindFirstChild("HumanoidRootPart")
		if not myRoot then return true end
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = { cachedChar, Workspace.CurrentCamera, part.Parent }
		return Workspace:Raycast(myRoot.Position, part.Position - myRoot.Position, params) ~= nil
	end

	local virtualUser
	function trigApi.fire(tool)
		if cfg.triggerClick then
			if type(mouse1click) == "function" then
				mouse1click()
				return
			end
			local c = Workspace.CurrentCamera
			if c then
				virtualUser = virtualUser or game:GetService("VirtualUser")
				virtualUser:CaptureController()
				virtualUser:Button1Down(Vector2.new(0, 0), c.CFrame)
				virtualUser:Button1Up(Vector2.new(0, 0), c.CFrame)
			end
			return
		end
		for _ = 1, cfg.shotsPerStep do
			pcall(function() tool:Activate() end)
		end
	end

	function getTool(char)
		if cTool and cTool.Parent == char then
			return cTool, cAmmo
		end
		cTool = char:FindFirstChildWhichIsA("Tool")
		cAmmo = cTool and cTool:FindFirstChild("Ammo")
		return cTool, cAmmo
	end

	function getTriggerDist(tool)
		if not cfg.perWeaponDist then
			return cfg.triggerMaxDist
		end
		return triggerDist[tool.Name] or triggerDist.Default
	end

	local function triggerStep()
		cachedChar = lp.Character
		if not cachedChar then return end
		
		cachedTool, cachedAmmo = getTool(cachedChar)
		if not cachedTool or not cachedAmmo or not triggerGuns[cachedTool.Name] then return end
		if cachedAmmo:IsA("IntValue") and cachedAmmo.Value <= 0 then return end
		
		local part = mouse.Target
		if not part or not part.Parent then return end
		
		local parent = part.Parent
		local target = Players:GetPlayerFromCharacter(parent)
			or (parent.Parent and Players:GetPlayerFromCharacter(parent.Parent))
		if not target or target == lp then return end
		if isIgnored(target) then return end
		
		local tchar = target.Character
		local hum = tchar and tchar:FindFirstChildOfClass("Humanoid")
		if not hum or hum.Health <= 0 then return end
		if trigApi.ko(target) then return end
		if tchar:FindFirstChild("GRABBING_CONSTRAINT") then return end
		
		local myRoot = cachedChar:FindFirstChild("HumanoidRootPart")
		if myRoot and (myRoot.Position - part.Position).Magnitude > getTriggerDist(cachedTool) then
			return
		end
		if cfg.triggerWallCheck and trigApi.wall(part) then return end
		
		if cfg.silentMethod == "Bullet teleport" and not cfg.tpTurbo and bulletTpShot(cachedTool, tchar) then
			return
		end
		trigApi.fire(cachedTool)
	end

	function updateTrigger()
		if triggerRunning then return end
		triggerRunning = true
		
		task.spawn(function()
			while triggerRunning do
				if isOn("trigger") and not (cfg.triggerRMB and not UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)) then
					pcall(triggerStep)
				end
				task.wait(0)
			end
		end)
		
		task.spawn(function()
			while triggerRunning do
				if isOn("trigger") and not (cfg.triggerRMB and not UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2)) then
					pcall(triggerStep)
				end
				task.wait(0)
			end
		end)
	end

	function disconnectTrigger()
		triggerRunning = false
	end
end -- triggerbot

local trigSilentRunning = false

local function trigSilentStep()
	local target = silentState.char
	if not target or not target.Parent then return end
	
	local myChar = lp.Character
	if not myChar then return end
	
	local tool, ammo = getTool(myChar)
	if not tool or not ammo or not triggerGuns[tool.Name] then return end
	if ammo:IsA("IntValue") and ammo.Value <= 0 then return end
	
	local myRoot = myChar:FindFirstChild("HumanoidRootPart")
	local targetRoot = target:FindFirstChild("HumanoidRootPart") or silentState.part
	if myRoot and targetRoot and (myRoot.Position - targetRoot.Position).Magnitude > getTriggerDist(tool) then
		return
	end
	
	local tplr = Players:GetPlayerFromCharacter(target)
	if tplr and isIgnored(tplr) then return end
	if tplr and trigApi.ko(tplr) then return end
	if target:FindFirstChild("GRABBING_CONSTRAINT") then return end
	if cfg.triggerWallCheck and silentState.part and trigApi.wall(silentState.part) then return end
	
	if cfg.silentMethod == "Bullet teleport" and not cfg.tpTurbo and bulletTpShot(tool, target) then
		return
	end
	pcall(trigApi.fire, tool)
end

---------------------------------------------------------------- Hitbox Expander (ОПТИМИЗИРОВАНО: обновление раз в 0.1с)
local hbCache = {}
local hbActiveKillers = {}
local warnedNoGetConnections = false
local hbLastUpdate = 0

local function warnNoExecutor()
	if warnedNoGetConnections then return end
	warnedNoGetConnections = true
end

local function isConnection(x)
	if x == nil then return false end
	if typeof(x) == "RBXScriptConnection" then return true end
	local ok = pcall(function() return x.Disconnect end)
	return ok
end

local function killConnection(conn)
	if not conn then return false end
	if pcall(function() conn:Disconnect() end) then return true end
	return pcall(function() conn:Disable() end)
end

local function nuke(sig)
	if not sig then return 0 end
	if type(getconnections) ~= "function" then warnNoExecutor() return 0 end
	local ok, conns = pcall(getconnections, sig)
	if not ok or not conns then return 0 end
	local killed = 0
	for _, conn in ipairs(conns) do
		if killConnection(conn) then killed = killed + 1 end
	end
	return killed
end

local HB_PROPERTIES = {
	"Size", "Transparency", "CanCollide", "CanQuery", "CanTouch",
	"Massless", "CFrame", "Position", "Color", "BrickColor",
	"Material", "Anchored", "Locked", "Reflectance", "Elasticity",
	"Friction", "Velocity", "RotVelocity", "AssemblyLinearVelocity",
	"AssemblyAngularVelocity"
}
local HB_EVENTS = {
	"ChildAdded", "ChildRemoved", "DescendantAdded", "DescendantRemoving",
	"AncestryChanged", "AttributeChanged", "Destroying", "Changed"
}

local function cleanSinglePart(hitbox)
	local total = 0
	for _, prop in ipairs(HB_PROPERTIES) do
		local ok, sig = pcall(function() return hitbox:GetPropertyChangedSignal(prop) end)
		if ok and sig then total = total + nuke(sig) end
	end
	local ok, changed = pcall(function() return hitbox.Changed end)
	if ok and changed then total = total + nuke(changed) end
	for _, evt in ipairs(HB_EVENTS) do
		local ok2, sig = pcall(function() return hitbox[evt] end)
		if ok2 and sig then total = total + nuke(sig) end
	end
	return total
end

local function killHitboxSignals(hitbox)
	if typeof(hitbox) ~= "Instance" or not hitbox:IsA("BasePart") then return 0 end
	return cleanSinglePart(hitbox)
end

local function startSignalKiller(player, hitbox)
	if not player or not player.UserId then return end
	if not hitbox or typeof(hitbox) ~= "Instance" then return end
	if hbActiveKillers[player.UserId] then return end
	if not player.Parent then return end

	hbActiveKillers[player.UserId] = {}
	local removing
	removing = Players.PlayerRemoving:Connect(function(p)
		if p == player then
			if hbActiveKillers[player.UserId] and hbActiveKillers[player.UserId].removing then
				pcall(function() hbActiveKillers[player.UserId].removing:Disconnect() end)
			end
			hbActiveKillers[player.UserId] = nil
		end
	end)
	hbActiveKillers[player.UserId].removing = removing

	task.spawn(function()
		local tries = 0
		while hbActiveKillers[player.UserId] and hitbox and hitbox.Parent and player.Parent and tries < 60 do
			killHitboxSignals(hitbox)
			task.wait(0.8)
			tries = tries + 1
		end
		if hbActiveKillers[player.UserId] and (not hitbox or not hitbox.Parent) then
			hbActiveKillers[player.UserId] = nil
		end
	end)
end

local function updateHitboxes()
	local now = tick()
	if now - hbLastUpdate < 0.1 then return end
	hbLastUpdate = now

	if not cfg.hitboxEnabled then
		for plr, hb in pairs(hbCache) do
			if hb and hb.Parent then
				hb.Size = Vector3.new(2, 2, 1)
				hb.Transparency = 1
			end
		end
		return
	end

	local myChar = lp.Character
	local hasWeapon = myChar and myChar:FindFirstChildOfClass("Tool") ~= nil
	
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr == lp then continue end
		if cfg.hitboxOnlyTarget and getRole(plr) ~= "Target" then continue end
		if getRole(plr) == "Friend" then continue end

		local char = plr.Character
		if not char then continue end
		
		local hb = char:FindFirstChild("Hitbox")
		if not hb or not hb:IsA("BasePart") then
			hbCache[plr] = nil
			continue
		end
		hbCache[plr] = hb

		if hasWeapon then
			if not hbActiveKillers[plr.UserId] then
				killHitboxSignals(hb)
				startSignalKiller(plr, hb)
			end
			local s = cfg.hitboxSize
			hb.Size = Vector3.new(s, s, s)
			hb.Transparency = cfg.hitboxTransparency
			hb.CanCollide = false
			hb.Massless = true
		else
			if hb.Size.X > 2 then hb.Size = Vector3.new(2, 2, 1) end
			hb.Transparency = 1
		end
	end
end

bind(RS.RenderStepped, function()
	updateHitboxes()
end)

local function autoShotSimulateClick()
	local delay = cfg.autoShotRandomDelay and (cfg.autoShotMinDelay + math.random() * (cfg.autoShotMaxDelay - cfg.autoShotMinDelay)) or cfg.autoShotDelayBeforeShot
	task.wait(delay)
	if type(mouse1press) == "function" and type(mouse1release) == "function" then
		pcall(function()
			mouse1press()
			task.wait(cfg.autoShotHoldTime)
			mouse1release()
		end)
	else
		local c = Workspace.CurrentCamera
		if c then
			local vu = game:GetService("VirtualUser")
			vu:CaptureController()
			vu:Button1Down(Vector2.new(0, 0), c.CFrame)
			task.wait(cfg.autoShotHoldTime)
			vu:Button1Up(Vector2.new(0, 0), c.CFrame)
		end
	end
end

local function autoShotDoBurst(shotsCount)
	if autoShotIsBursting then return end
	autoShotIsBursting = true
	task.spawn(function()
		for i = 1, shotsCount do
			autoShotSimulateClick()
			if i < shotsCount then task.wait(cfg.autoShotBurstDelay) end
		end
		autoShotIsBursting = false
	end)
end

local function autoShotCheck()
	if not cfg.autoShotEnabled then return end
	if #autoShotHosts == 0 then return end
	
	for _, host in ipairs(autoShotHosts) do
		if not host or isIgnored(host) then continue end
		local char = host.Character
		if not char then continue end
		local tool = char:FindFirstChildWhichIsA("Tool")
		if not tool or not cfg.autoShotWeapons[tool.Name] then continue end
		
		local ammo = tool:FindFirstChild("Ammo")
		if not ammo or not ammo:IsA("IntValue") then continue end
		
		local curr = ammo.Value
		local prev = autoShotLastAmmo[tool]
		if prev ~= nil and curr < prev then
			if cfg.autoShotBurstShots > 1 then
				autoShotDoBurst(cfg.autoShotBurstShots)
			else
				autoShotSimulateClick()
			end
		end
		autoShotLastAmmo[tool] = curr
	end
end

bind(RS.Heartbeat, autoShotCheck)

local function autoShotAddHost(plr)
	if #autoShotHosts >= 3 then return false end
	for _, p in ipairs(autoShotHosts) do if p == plr then return false end end
	table.insert(autoShotHosts, plr)
	autoShotAttachEmoji(plr)
	return true
end

local function autoShotRemoveHost(plr)
	for i, p in ipairs(autoShotHosts) do
		if p == plr then
			table.remove(autoShotHosts, i)
			autoShotRemoveEmoji(plr)
			break
		end
	end
end

local function autoShotIsHost(plr)
	for _, p in ipairs(autoShotHosts) do if p == plr then return true end end
	return false
end

local function autoShotGetClosestToCrosshair()
	local mouseLoc = UIS:GetMouseLocation()
	local closest, dist = nil, math.huge
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= lp and plr.Character and plr.Character:FindFirstChild("Head") then
			local head = plr.Character.Head
			local sp, onScreen = cam:WorldToViewportPoint(head.Position)
			if onScreen then
				local d = (Vector2.new(sp.X, sp.Y) - mouseLoc).Magnitude
				if d < dist then dist = d; closest = plr end
			end
		end
	end
	return closest
end

			line.AnchorPoint = Vector2.new(0.5, 0.5)
			line.BackgroundColor3 = Color3.new(1, 1, 1)
			line.BorderSizePixel = 0
			line.Visible = false
			line.Parent = menuFxOverlay
			menuFxLines[i] = line
		end
		return line
	end

	local function menuFxShow()
		if not menuFxOverlay then return end
		menuFxShown = true
		TweenService:Create(menuFxOverlay, TweenInfo.new(MENU_FX_FADE), {
			BackgroundTransparency = 1 - math.clamp(cfg.menuFxDim, 0, 90) / 100,
		}):Play()
		if menuFxBlur then
			TweenService:Create(menuFxBlur, TweenInfo.new(MENU_FX_FADE), {
				Size = math.clamp(cfg.menuFxBlur, 0, 40),
			}):Play()
		end
	end

	local function menuFxHide()
		if not menuFxOverlay then return end
		menuFxShown = false
		for _, p in ipairs(menuFxParticles) do p.obj.Visible = false end
		for _, l in ipairs(menuFxLines) do l.Visible = false end
		menuFxVisible = false
		TweenService:Create(menuFxOverlay, TweenInfo.new(MENU_FX_FADE), { BackgroundTransparency = 1 }):Play()
		if menuFxBlur then
			TweenService:Create(menuFxBlur, TweenInfo.new(MENU_FX_FADE), { Size = 0 }):Play()
		end
	end

	function menuFxSync(state)
		if not menuFxOverlay then return end
		if state and cfg.menuFx then
			menuFxShow()
		else
			menuFxHide()
		end
	end

	bind(RS.RenderStepped, function(dt)
		if not menuFxOverlay then return end
		if not menuFxShown and menuFxOverlay.BackgroundTransparency >= 0.99 then
			if menuFxVisible then
				for _, p in ipairs(menuFxParticles) do p.obj.Visible = false end
				for _, l in ipairs(menuFxLines) do l.Visible = false end
				menuFxVisible = false
			end
			return
		end
		menuFxVisible = true
		if not cfg.menuFxParticles then
			for _, p in ipairs(menuFxParticles) do p.obj.Visible = false end
			for _, l in ipairs(menuFxLines) do l.Visible = false end
			return
		end
		local vp = menuFxViewport()
		local mouse = UIS:GetMouseLocation()
		local radius = cfg.menuFxLink
		local used = 0
		for _, p in ipairs(menuFxParticles) do
			p.y = p.y + p.speed * dt
			p.x = p.x + p.drift * dt
			if p.y > vp.Y or p.x > vp.X or p.x < 0 then
				p.y = -10
				p.x = menuFxRng:NextNumber(0, vp.X)
			end
			p.obj.Visible = true
			p.obj.Position = UDim2.fromOffset(p.x, p.y)
			local dx, dy = p.x - mouse.X, p.y - mouse.Y
			local dist = math.sqrt(dx * dx + dy * dy)
			if radius > 0 and dist < radius then
				used = used + 1
				local line = menuFxLine(used)
				line.Visible = true
				line.Size = UDim2.fromOffset(dist, 1)
				line.Position = UDim2.fromOffset((p.x + mouse.X) / 2, (p.y + mouse.Y) / 2)
				line.Rotation = math.deg(math.atan2(dy, dx))
				line.BackgroundTransparency = dist / radius
			end
		end
		for i = used + 1, #menuFxLines do
			menuFxLines[i].Visible = false
		end
	end)
end -- Menu background

setMenu(cfg.menuOpen)

bind(UIS.InputBegan, function(inp, processed)
	if processed then return end
	
	if inp.KeyCode == Enum.KeyCode.Delete then
		setMenu(not cfg.menuOpen)
		return
	end

	if inp.KeyCode == cfg.panicKey then
		pcall(env.StarTurbUnload)
		pcall(function()
			if Library and Library.Unload then Library:Unload() end
		end)
	end

	if inp.KeyCode == Enum.KeyCode.F1 then autoShotF1Down = true end
	if inp.KeyCode == Enum.KeyCode.Z then autoShotZDown = true end
	if inp.KeyCode == Enum.KeyCode.K then autoShotKDown = true end
	
	if autoShotF1Down and autoShotZDown then
		local closest = autoShotGetClosestToCrosshair()
		if closest then
			if autoShotIsHost(closest) then
				autoShotRemoveHost(closest)
				autoShotLastAmmo = {}
				notify("Host removed: " .. closest.Name)
			else
				if cfg.autoShotAllowFriendsAsHost or getRole(closest) ~= "Friend" then
					if autoShotAddHost(closest) then 
						autoShotLastAmmo = {}
						notify("Host added: " .. closest.Name)
					end
				end
			end
		end
	end
	
	if autoShotF1Down and autoShotKDown then
		cfg.autoShotShowEmoji = not cfg.autoShotShowEmoji
		if cfg.autoShotShowEmoji then
			for _, plr in ipairs(autoShotHosts) do autoShotAttachEmoji(plr) end
		else
			for _, emoji in pairs(autoShotEmojis) do pcall(function() emoji:Destroy() end) end
			autoShotEmojis = {}
		end
		notify("Emoji: " .. (cfg.autoShotShowEmoji and "ON" or "OFF"))
	end
end)

bind(UIS.InputEnded, function(inp, processed)
	if inp.KeyCode == Enum.KeyCode.F1 then autoShotF1Down = false end
	if inp.KeyCode == Enum.KeyCode.Z then autoShotZDown = false end
	if inp.KeyCode == Enum.KeyCode.K then autoShotKDown = false end
end)

env.StarTurbUnload = function()
	cfg.trigger, cfg.rapid, cfg.silent, cfg.trigSilent, cfg.hitboxEnabled, cfg.autoShotEnabled = false, false, false, false, false, false

	pcall(mdCleanup)
	pcall(visClear)
	pcall(silentDisableHooks)
	pcall(tpRestore)
	pcall(disconnectTrigger)
	pcall(disconnectRapid)

	for _, emoji in pairs(autoShotEmojis) do pcall(function() emoji:Destroy() end) end
	autoShotEmojis = {}

	freeDrawing(silentCircle)
	if friendsGui then pcall(function() friendsGui:Destroy() end) end
	if menuFxBlur then pcall(function() menuFxBlur:Destroy() end) end
	if menuFxGui then pcall(function() menuFxGui:Destroy() end) end

	for _, c in ipairs(conns) do pcall(function() c:Disconnect() end) end
	conns = {}

	pcall(function()
		if Library and Library.Unload then Library:Unload() end
	end)

	env.StarTurbUnload = nil
end
